SQLite 3 Embedded Database & CLI

Complete relational SQL database engine compiled to WebAssembly with in-memory tables and persistent virtual filesystem storage.

Database Storage Modes

  • In-Memory Mode (:memory:): Ultra-fast transient RAM database. Data is wiped when the process exits. 
    sqlite3_open(":memory:", &db);
  • Persistent VFS File Mode (/workspace/*.db): Stores data inside the in-memory VFS under /workspace. IndexedDB automatically persists changes across browser reloads. 
    sqlite3_open("/workspace/app.db", &db);

Core C Program Example

#include <stdio.h>
#include <sqlite3.h>

int main(void) {
    sqlite3 *db;
    if (sqlite3_open(":memory:", &db) != SQLITE_OK) {
        fprintf(stderr, "Cannot open database: %s\n", sqlite3_errmsg(db));
        return 1;
    }

    char *err_msg = NULL;
    const char *sql = "CREATE TABLE students (id INT, name TEXT, score REAL);"
                      "INSERT INTO students VALUES (1, 'Alice', 94.5);"
                      "INSERT INTO students VALUES (2, 'Bob', 88.0);";

    if (sqlite3_exec(db, sql, 0, 0, &err_msg) != SQLITE_OK) {
        fprintf(stderr, "SQL error: %s\n", err_msg);
        sqlite3_free(err_msg);
        sqlite3_close(db);
        return 1;
    }

    sqlite3_stmt *stmt;
    sqlite3_prepare_v2(db, "SELECT id, name, score FROM students", -1, &stmt, NULL);

    printf("%-4s %-10s %-6s\n", "ID", "Name", "Score");
    while (sqlite3_step(stmt) == SQLITE_ROW) {
        printf("%-4d %-10s %-6.1f\n",
            sqlite3_column_int(stmt, 0),
            sqlite3_column_text(stmt, 1),
            sqlite3_column_double(stmt, 2));
    }

    sqlite3_finalize(stmt);
    sqlite3_close(db);
    return 0;
}

Interactive SQLite3 CLI Shell

Run the genuine SQLite CLI applet directly in the terminal to inspect and query database files:

$ sqlite3 /workspace/mydb.db
SQLite version 3.54.1
Enter ".help" for usage hints.
sqlite> CREATE TABLE items (id INTEGER PRIMARY KEY, item TEXT);
sqlite> INSERT INTO items (item) VALUES ('Keyboard'), ('Mouse');
sqlite> SELECT * FROM items;
1|Keyboard
2|Mouse
sqlite> .tables
items
sqlite> .quit
